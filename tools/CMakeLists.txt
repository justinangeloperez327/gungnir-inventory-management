cmake_minimum_required(VERSION 3.25)
project(inventory_provision LANGUAGES CXX)
find_package(Gungnir CONFIG REQUIRED)
if(NOT Gungnir_WITH_SQLITE OR NOT Gungnir_WITH_PASSWORD)
    message(FATAL_ERROR "The Gungnir Application SDK (SQLite + password) is required")
endif()
add_executable(inventory-provision-user provision_user.cpp)
target_compile_features(inventory-provision-user PRIVATE cxx_std_23)
target_link_libraries(inventory-provision-user PRIVATE gungnir::gungnir gungnir::orm)
if(MSVC)
    set_property(TARGET inventory-provision-user PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
